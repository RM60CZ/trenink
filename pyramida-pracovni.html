python def vytvor_pyramidu(): max_radku = 7 sirka_pole = 6 # pro čísla jako "12 345" celkova_sirka = max_radku * sirka_pole for radek in range(1, max_radku + 1): # Počet čtverečků na aktuálním řádku pocet_ctvercu = radek # Vytvoření jednoho řádku: spojení 'pocet_ctvercu' krát '[ ]' radek_text = ''.join(f"[{' ':>{sirka_pole - 2}}]" for _ in range(pocet_ctvercu)) # Výpočet odsazení pro zarovnání na střed mezery = (celkova_sirka - len(radek_text)) // 2 print(' ' * mezery + radek_text) # Spuštění vytvor_pyramidu()
